extends SceneTree
## One-off headless sanity for junction and side-street facade spans (not part of the game).


func _initialize() -> void:
	call_deferred(&"_run")


func _run() -> void:
	await process_frame
	var failures := 0
	for scene_path: String in [
		"res://scenes/corridor/corridor_t_junction.tscn",
		"res://scenes/corridor/corridor_crossroads.tscn",
	]:
		var packed: PackedScene = load(scene_path)
		for d in 5:
			var junction: Node3D = packed.instantiate()
			root.add_child(junction)
			await process_frame
			junction.configure(5 + d, d, 9)
			var facades: Node3D = junction.get_node("Facades")
			var names: Array[String] = []
			for child in facades.get_children():
				names.append(child.name)
			var meshes := facades.find_children("*", "MeshInstance3D", true, false)
			print("%s district %d: spans=%s meshes=%d" % [scene_path.get_file(), d, names, meshes.size()])
			if meshes.size() < 8:
				failures += 1
				print("FAIL: too few meshes")
			junction.queue_free()
			await process_frame
	var branch_packed: PackedScene = load("res://scenes/corridor/side_street_branch.tscn")
	for mirrored in [false, true]:
		var branch: Node3D = branch_packed.instantiate()
		branch.mirror_x = mirrored
		root.add_child(branch)
		await process_frame
		branch.configure(3, 1, 9)
		var far: Node3D = branch.get_node("Facades/FarEnd")
		var body: MeshInstance3D = far.find_children("Body*", "MeshInstance3D", true, false)[0]
		var aabb: AABB = body.global_transform * body.get_aabb()
		print("branch mirror=%s far-end body x %.2f..%.2f z %.2f..%.2f y %.2f..%.2f" % [
			mirrored, aabb.position.x, aabb.end.x, aabb.position.z, aabb.end.z,
			aabb.position.y, aabb.end.y
		])
		var north: Node3D = branch.get_node("Facades/FlankNorth")
		var nb: MeshInstance3D = north.find_children("Body*", "MeshInstance3D", true, false)[0]
		var naabb: AABB = nb.global_transform * nb.get_aabb()
		print("  flank north body x %.2f..%.2f z %.2f..%.2f" % [
			naabb.position.x, naabb.end.x, naabb.position.z, naabb.end.z
		])
		if mirrored and aabb.position.x < 0.0:
			failures += 1
			print("FAIL: mirrored far end is not at +x")
		if not mirrored and aabb.end.x > 0.0:
			failures += 1
			print("FAIL: unmirrored far end is not at -x")
		branch.queue_free()
		await process_frame
	print("SPAN SANITY %s (%d failures)" % ["OK" if failures == 0 else "FAILED", failures])
	quit(0 if failures == 0 else 1)
